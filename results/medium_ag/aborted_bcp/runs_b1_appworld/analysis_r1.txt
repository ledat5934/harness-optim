"""context_mgmt/default: system prompt + guidance + memory notes, pinned task message, sliding window.

Keeps the first user message (task) pinned and the most recent `keep_turns` assistant turns, together with the
tool/user replies that belong to them, so native tool-call pairing is never broken. When older turns are
dropped, a one-line marker says how many were elided. Each tool observation is also capped at `max_obs_chars`
inside the context. This matches the simplified AppWorld agent's `trimmed_messages` (drop the oldest
non-instruction messages when too long) and MemoHarness's stabilized default (sliding window that pins the
instruction).
"""
from __future__ import annotations

_DEFAULT_GUIDANCE = (
    "Operational Guidelines:\n"
    "1. Authentication & Access Tokens: App APIs are stateless REST endpoints. Calling `apis.<app>.login(...)` "
    "returns a dictionary containing `access_token`. You must explicitly pass `access_token=...` to all subsequent "
    "API calls that require authentication.\n"
    "2. Account Passwords: User passwords for apps and accounts are stored in the supervisor app. Retrieve them "
    "using `apis.supervisor.show_account_passwords()`.\n"
    "3. API Discovery: Inspect API schemas and required arguments using `apis.api_docs.show_api_doc(app_name, api_name)` "
    "or `apis.api_docs.show_api_descriptions(app_name)` before calling unfamiliar endpoints.\n"
    "4. Pagination: Endpoints returning lists of items support pagination (`page_index`, `page_limit`). Paginate through "
    "all pages in a loop to ensure you gather all items without truncation.\n"
    "5. Efficient Execution: Write consolidated, multi-statement Python code in `execute_python` to perform login, "
    "discovery, data retrieval, and operations in cohesive steps rather than issuing one single-line command per turn. "
    "This conserves the limited step budget.\n"
    "6. Completion: When the task is accomplished, verify the final state and call `apis.supervisor.complete_task(answer=...)`. "
    "Only pass `answer` if the instruction explicitly asks a question requiring an answer."
)


class SlidingWindow:
    def __init__(self, keep_turns: int = 30, max_obs_chars: int = 6000, extra_system: str = _DEFAULT_GUIDANCE):
        self.keep_turns, self.max_obs_chars, self.extra_system = keep_turns, max_obs_chars, extra_system

    def build(self, system, task, history, memory_notes):
        sys_parts = [system]
        if task.guidance:
            sys_parts.append(task.guidance)
        if self.extra_system:
            sys_parts.append(self.extra_system)
        if memory_notes:
            sys_parts.append("Notes from previous experience:\n" + memory_notes)
        msgs = [{"role": "system", "content": "\n\n".join(p for p in sys_parts if p)}]
        if not history:
            return msgs
        head, rest = history[0], history[1:]
        # group: an assistant message followed by its tool/user replies
        groups, cur = [], []
        for m in rest:
            if m["role"] == "assistant" and cur:
                groups.append(cur)
                cur = []
            cur.append(m)
        if cur:
            groups.append(cur)
        dropped = max(0, len(groups) - self.keep_turns)
        kept = groups[dropped:]
        msgs.append(head)
        if dropped:
            msgs.append({"role": "user", "content": f"[{dropped} earlier steps omitted from context]"})
        for g in kept:
            for m in g:
                if m["role"] == "tool" and len(m.get("content") or "") > self.max_obs_chars:
                    c = m["content"]
                    m = {**m, "content": c[: self.max_obs_chars] + f"\n...[truncated {len(c) - self.max_obs_chars} chars]"}
                msgs.append(m)
        return msgs


def register(reg):
    reg.register("context_mgmt", "default", lambda c: SlidingWindow(**c),
                 "System prompt + benchmark guidance + operational guidelines + memory notes; task message pinned; "
                 "sliding window over the last 30 assistant turns (tool pairs kept intact); observations capped at 6000 chars.",
                 config_schema={"keep_turns": "int (default 30)", "max_obs_chars": "int (default 6000)",
                                "extra_system": f"str appended to the system prompt (default operational guidelines)"},
                 niche={"strategy": "sliding_window", "summarize": "no"})
